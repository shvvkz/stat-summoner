17/// ⚙️ **Function**: Checks the database for followed summoners and updates their information from the Riot API.
18///
19/// This asynchronous function queries the MongoDB collection to check if there are any followed summoners.
20/// If documents are present, it retrieves the followed summoners and processes each one by fetching the latest data from the Riot API.
21/// The function is designed to keep the database in sync with real-time summoner information.
22///
23/// # Parameters:
24/// - `mongo_client`: A reference to the MongoDB `Client`, used to query and update the database.
25/// - `riot_api_key`: A string slice representing the Riot API key, required to make authorized API calls.
26/// - `http`: An `Arc<Http>` reference to the HTTP client used for making requests to the Riot API.
27///
28/// # Returns:
29/// - `Result<(), Error>`: Returns an empty result if successful, or an error if any part of the process fails.
30///
31/// # Example:
32/// This function is used to periodically check and update summoner information.
33///
34/// ```rust
35/// check_and_update_db(&mongo_client, riot_api_key, http.clone()).await?;
36/// ```
37///
38/// # Notes:
39/// - The function first checks if there are any documents in the `follower_summoner` collection. If the collection is empty, no further action is taken.
40/// - For each followed summoner, the function retrieves their latest match data using the Riot API and updates the database accordingly.
41pub async fn check_and_update_db(
42    mongo_client: &Client,
43    riot_api_key: &str,
44    http: Arc<Http>,
45) -> Result<(), Error> {
46    let collection = mongo_client
47        .database("stat-summoner")
48        .collection::<SummonerFollowedData>("follower_summoner");
49    let collection_emoji = mongo_client
50        .database("stat-summoner")
51        .collection::<EmojiId>("emojis_id");
52    let count = collection.estimated_document_count().await?;
53
54    if count > 0 {
55        let followed_summoners = get_followed_summoners(&collection).await?;
56        for followed_summoner in followed_summoners {
57            process_followed_summoner(
58                &collection,
59                &followed_summoner,
60                riot_api_key,
61                http.clone(),
62                collection_emoji.clone(),
63            )
64            .await?;
65        }
66    }
67
68    Ok(())
69}

71/// ⚙️ **Function**: Fetches champion data from League of Graphs and updates MongoDB.
72///
73/// This asynchronous function retrieves champion statistics, rune data, and core build information
74/// from the League of Graphs website. It processes the HTML content to extract data for each champion
75/// and updates or inserts this information into the MongoDB database. If the champion already exists in the database,
76/// it updates their data; otherwise, it inserts a new document.
77///
78/// # Parameters:
79/// - `mongo_client`: A reference to the MongoDB `Client`, used to query and update the MongoDB database.
80///
81/// # Returns:
82/// - `Result<(), Box<dyn std::error::Error>>`: Returns an empty result if successful, or an error if any part of the process fails.
83///
84/// # Example:
85/// This function is typically called to fetch and update champion data in a scheduled task:
86///
87/// ```rust
88/// fetch_champion_data(&mongo_client).await?;
89/// ```
90///
91/// # Notes:
92/// - The function starts by sending an HTTP request to the League of Graphs page to fetch champion build data.
93/// - It parses the HTML content using the `select` crate, extracting details such as popularity, win rate, and ban rate for each champion.
94/// - For each champion, it also retrieves runes and core build information using the `fetch_runes` and `fetch_core_build` functions.
95/// - The MongoDB collection `champions_data` is then updated with the latest data for each champion. If the champion already exists, the data is updated; otherwise, a new entry is inserted.
96/// - The function makes use of `task::spawn_blocking` to handle blocking operations during HTML parsing.
97pub async fn fetch_champion_data(mongo_client: &Client) -> Result<(), Box<dyn std::error::Error>> {
98    let url = "https://www.leagueofgraphs.com/champions/builds";
99    let dd_json = open_dd_json().await.unwrap();
100    let client = reqwest::Client::new();
101    let res = client
102        .get(url)
103        .header("User-Agent", "Mozilla/5.0")
104        .send()
105        .await?;
106
107    let body = res.text().await?;
108
109    // 🔽 Ajoute ce bloc pour afficher les scripts contenant "$(function"
110    std::fs::write("/tmp/champions_builds.html", &body)?;
111
112    let results: Vec<ChampionData> = task::spawn_blocking(move || {
113        let mut results = Vec::new();
114        let document = select::document::Document::from(body.as_str());
115        for node in document.find(select::predicate::Name("script")) {
116            let text = node.text();
117
118            if text.contains("$(function () {\n        ChampionsPage.init") {
119                let lines: Vec<&str> = text.lines().collect();
120                let data = if lines.len() >= 3 {
121                    extract_info_from_json_arr(lines[2], "rankings")
122                } else {
123                    vec![]
124                };
125
126                // Debug d’affichage
127                for c in data.iter() {
128                    let default_runes = RunesData {
129                        parent_primary_rune: String::new(),
130                        child_primary_rune_1: String::new(),
131                        child_primary_rune_2: String::new(),
132                        child_primary_rune_3: String::new(),
133                        child_secondary_rune_1: String::new(),
134                        child_secondary_rune_2: String::new(),
135                        tertiary_rune_1: String::new(),
136                        tertiary_rune_2: String::new(),
137                        tertiary_rune_3: String::new(),
138                    };
139                    let default_core_build = CoreBuildData {
140                        first: String::new(),
141                        second: String::new(),
142                        third: String::new(),
143                    };
144                    let runes = block_on(fetch_runes(&c.champion_link)).unwrap_or(default_runes);
145
146                    let core_build =
147                        block_on(fetch_core_build(&c.champion_link)).unwrap_or(default_core_build);
148                    let id_name = dd_json["data"]
149                        .as_object()
150                        .and_then(|data| {
151                            data.values().find(|champion| {
152                                champion["name"]
153                                    .as_str()
154                                    .map_or(false, |n| n == c.champion_name)
155                            })
156                        })
157                        .and_then(|champion: &serde_json::Value| champion["id"].as_str())
158                        .unwrap_or(&c.champion_name)
159                        .to_string();
160                    results.push(ChampionData {
161                        name: c.champion_name.clone(),
162                        role: c.roles.clone(),
163                        popularity: c.popularity_played_percentage.clone(),
164                        winrate: c.popularity_winrate.clone(),
165                        banrate: c.ban_rate.clone(),
166                        id_name: id_name.clone(),
167                        runes,
168                        core_build,
169                    });
170                }
171            }
172        }
173        results
174    })
175    .await?;
176
177    let collection = mongo_client
178        .database("stat-summoner")
179        .collection::<ChampionData>("champions_data");
180
181    for champion in results {
182        let filter = doc! { "name": &champion.name };
183
184        if let Some(_) = collection.find_one(filter.clone()).await? {
185            let update = doc! {
186                "$set": {
187                    "role": champion.role,
188                    "popularity": champion.popularity,
189                    "winrate": champion.winrate,
190                    "banrate": champion.banrate,
191                    "id_name": champion.id_name,
192                    "runes": bson::to_document(&champion.runes).unwrap(),
193                    "core_build":  bson::to_document(&champion.core_build).unwrap()
194                }
195            };
196            collection.update_one(filter, update).await?;
197        } else {
198            collection.insert_one(champion).await?;
199        }
200    }
201
202    log::info!("Mise à jour des données MongoDB terminée.");
203    Ok(())
204}
