15/// Fetches and displays LoL player stats based on user input.
16///
17/// This Discord command allows a user to input their League of Legends in-game name and tag, then fetches
18/// the player's Solo/Duo and Flex rank, top champions, and recent match details from the Riot API.
19/// The results are displayed in a formatted embed and automatically deleted after 60 seconds.
20///
21/// # Parameters:
22/// - `ctx`: The application context, providing access to Discord interaction methods and the Riot API key.
23/// - `region`: The region selected by the user (e.g., `Region::EUW`, `Region::NA`) to fetch statistics from the appropriate server.
24///
25/// # Returns:
26/// - `Result<(), Error>`: If successful, returns `Ok(())`, otherwise returns an error.
27///
28/// # ⚠️ Notes:
29/// - The command opens a modal dialog to gather the player's in-game name and tag.
30/// - The message displaying the player's stats is automatically deleted after 60 seconds to keep the chat clean.
31///
32/// # Example:
33/// ```rust
34/// lolstats(ctx, Region::EUW).await?;
35/// ```
36///
37/// This command displays information such as:
38/// ```text
39/// 📊 Stats for Faker#1234
40/// 🔱 **Solo/Duo Rank**: Gold I (100 LP)
41/// 🌀 **Flex Rank**: Silver IV (50 LP)
42/// 💥 **Top Champions**:
43/// Yasuo - Level: 7 - Points: 123456
44/// 📜 **Match Details**:
45/// Victory - **Yasuo**, 2 hours ago (Ranked Solo/Duo):
46/// K/D/A: **10/2/8** | **200 CS** | Duration: **30:45**
47/// ⏳ Played: **2 hours ago**
48/// ```
49#[poise::command(slash_command)]
50pub async fn lolstats(
51    ctx: poise::ApplicationContext<'_, Data, Error>,
52    #[description = "Select your region"] region: Region,
53) -> Result<(), Error> {
54    let modal_data: LolStatsModal = match LolStatsModal::execute(ctx).await {
55        Ok(Some(data)) => data,
56        Ok(None) => {
57            let error_message = "Modal data not found.";
58            let reply = ctx.send(create_embed_error(&error_message)).await?;
59            schedule_message_deletion(reply, ctx).await?;
60            return Ok(());
61        }
62        Err(_) => {
63            let error_message = "Failed to retrieve modal data.";
64            let reply = ctx.send(create_embed_error(&error_message)).await?;
65            schedule_message_deletion(reply, ctx).await?;
66            return Ok(());
67        }
68    };
69
70    manage_user(
71        ctx.author().id.to_string(),
72        ctx.author().name.clone(),
73        &ctx.data().mongo_client,
74        false,
75    )
76    .await?;
77
78    let client = Client::new();
79    let game_name_space = modal_data.game_name.replace(" ", "%20");
80
81    let region_str = region_to_string(&region);
82
83    let puuid = match get_puuid(
84        &client,
85        &game_name_space,
86        &modal_data.tag_line,
87        &ctx.data().riot_api_key,
88    )
89    .await
90    {
91        Ok(puuid) => puuid,
92        Err(e) => {
93            let error_message = format!("Error fetching PUUID: {}", e);
94            let reply = ctx.send(create_embed_error(&error_message)).await?;
95            schedule_message_deletion(reply, ctx).await?;
96            return Ok(());
97        }
98    };
99    // println!("PUUID: {}", puuid);
100    // let summoner_id =
101    //     match get_summoner_id(&client, &region_str, &puuid, &ctx.data().riot_api_key).await {
102    //         Ok(id) => id,
103    //         Err(e) => {
104    //             let error_message = format!("Error fetching summoner ID: {}", e);
105    //             let reply = ctx.send(create_embed_error(&error_message)).await?;
106    //             schedule_message_deletion(reply, ctx).await?;
107    //             return Ok(());
108    //         }
109    //     };
110
111    let (rank_info_res, champions_res, match_ids_res) = join!(
112        get_rank_info(&client, &region_str, &puuid, &ctx.data().riot_api_key),
113        get_champions(&client, &puuid, &region_str, &ctx.data().riot_api_key),
114        get_matchs_id(&client, &puuid, &ctx.data().riot_api_key, 5)
115    );
116
117    let rank_info = match rank_info_res {
118        Ok(info) => info,
119        Err(e) => {
120            let error_message = format!("Error fetching rank info: {}", e);
121            let reply = ctx.send(create_embed_error(&error_message)).await?;
122            schedule_message_deletion(reply, ctx).await?;
123            return Ok(());
124        }
125    };
126
127    let champions = match champions_res {
128        Ok(champs) => champs,
129        Err(e) => {
130            let error_message = format!("Error fetching champions: {}", e);
131            let reply = ctx.send(create_embed_error(&error_message)).await?;
132            schedule_message_deletion(reply, ctx).await?;
133            return Ok(());
134        }
135    };
136
137    let match_ids = match match_ids_res {
138        Ok(ids) => ids,
139        Err(e) => {
140            let error_message = format!("Error fetching match IDs: {}", e);
141            let reply = ctx.send(create_embed_error(&error_message)).await?;
142            schedule_message_deletion(reply, ctx).await?;
143            return Ok(()); // Retourne Ok(()) pour terminer proprement
144        }
145    };
146
147    let mut default_rank = HashMap::new();
148    default_rank.insert(
149        "tier".to_string(),
150        serde_json::Value::String("Unranked".to_string()),
151    );
152    default_rank.insert(
153        "rank".to_string(),
154        serde_json::Value::String("".to_string()),
155    );
156    default_rank.insert(
157        "leaguePoints".to_string(),
158        serde_json::Value::Number(0.into()),
159    );
160    default_rank.insert("wins".to_string(), serde_json::Value::Number(0.into()));
161    default_rank.insert("losses".to_string(), serde_json::Value::Number(0.into()));
162    default_rank.insert(
163        "queueType".to_string(),
164        serde_json::Value::String("".to_string()),
165    );
166
167    let (solo_rank, flex_rank) = determine_solo_flex(&rank_info, &default_rank);
168    let mongo_client: &mongodb::Client = &ctx.data().mongo_client;
169    let collection_emoji = mongo_client
170        .database("stat-summoner")
171        .collection::<EmojiId>("emojis_id");
172    let reply = create_and_send_embed_lolstats(
173        &modal_data,
174        puuid,
175        &solo_rank,
176        &flex_rank,
177        champions,
178        match_ids,
179        &ctx,
180        collection_emoji,
181    )
182    .await;
183    let sent_message = ctx.send(reply).await?;
184    if let Err(e) = schedule_message_deletion(sent_message, ctx).await {
185        log::error!("Failed to schedule message deletion: {}", e);
186    }
187    Ok(())
188}
