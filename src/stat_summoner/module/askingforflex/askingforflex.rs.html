9use poise::Modal;

11/// Handles the flex asking command.
12///
13/// This command updates or creates the user in the database, retrieves the emoji collection,
14/// fetches the guild roles, executes the modal, validates the data, generates the role mention,
15/// composes the flex request message, creates the embed, and sends the message with buttons for
16/// other users to join.
17///
18/// # Arguments
19///
20/// * `ctx` - The application context containing data and error types.
21///
22/// # Returns
23///
24/// * `Result<(), Error>` - Returns an empty result on success or an error on failure.
25#[poise::command(slash_command)]
26pub async fn askingforflex(ctx: poise::ApplicationContext<'_, Data, Error>) -> Result<(), Error> {
27    manage_user(
28        ctx.author().id.to_string(),
29        ctx.author().name.clone(),
30        &ctx.data().mongo_client,
31        false,
32    )
33    .await?;
34
35    let collection_emojis = ctx
36        .data()
37        .mongo_client
38        .database("stat-summoner")
39        .collection::<EmojiId>("emojis_id");
40
41    let guild_roles = ctx.guild().map(|g| g.roles.clone());
42
43    let raw_modal = FlexAskingModal::execute(ctx).await?;
44    let modal_data = match parse_flex_modal(ctx, raw_modal).await {
45        Some(data) => data,
46        None => return Ok(()),
47    };
48
49    let role_mention = compute_role_mention(guild_roles, modal_data.people_to_ask_by_role.trim());
50
51    let flex_message = format!(
52        "Hey {} please wake up! {} is looking for a flex game at **{}**!",
53        role_mention,
54        ctx.author().mention(),
55        modal_data.starting_hour
56    );
57
58    let embed = create_flex_embed(&ctx.author().name, &modal_data, collection_emojis).await?;
59
60    let button_top = poise::serenity_prelude::CreateButton::new("Top")
61        .custom_id("flex_user:top")
62        .label("Top")
63        .style(poise::serenity_prelude::ButtonStyle::Primary);
64
65    let button_jungle = poise::serenity_prelude::CreateButton::new("Jungle")
66        .custom_id("flex_user:jungle")
67        .label("Jungle")
68        .style(poise::serenity_prelude::ButtonStyle::Primary);
69
70    let button_mid = poise::serenity_prelude::CreateButton::new("Mid")
71        .custom_id("flex_user:mid")
72        .label("Mid")
73        .style(poise::serenity_prelude::ButtonStyle::Primary);
74
75    let button_adc = poise::serenity_prelude::CreateButton::new("ADC")
76        .custom_id("flex_user:adc")
77        .label("ADCarry")
78        .style(poise::serenity_prelude::ButtonStyle::Primary);
79
80    let button_support = poise::serenity_prelude::CreateButton::new("Support")
81        .custom_id("flex_user:support")
82        .label("Support")
83        .style(poise::serenity_prelude::ButtonStyle::Primary);
84
85    let builder = CreateMessage::new()
86        .content(flex_message)
87        .embed(embed)
88        .button(button_top)
89        .button(button_jungle)
90        .button(button_mid)
91        .button(button_adc)
92        .button(button_support);
93    ctx.channel_id().send_message(&ctx.http(), builder).await?;
94
95    Ok(())
96}
