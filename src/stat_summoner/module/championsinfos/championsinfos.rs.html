12/// Fetches and displays detailed information about a League of Legends champion based on user input.
13///
14/// This Discord command allows a user to input the name of a League of Legends champion.
15/// It then fetches the champion's data from the database, including roles, winrate, banrate, popularity,
16/// recommended runes, and core item build. The information is displayed in a formatted embed with appropriate emojis,
17/// and the message is automatically deleted after a certain period to keep the chat clean.
18///
19/// # Parameters:
20/// - `ctx`: The application context, providing access to Discord interaction methods, data dragon JSON, and the MongoDB client.
21///
22/// # Returns:
23/// - `Result<(), Error>`: If successful, returns `Ok(())`; otherwise, returns an error.
24///
25/// # ⚠️ Notes:
26/// - The command opens a modal dialog to collect the champion's name from the user.
27/// - It uses fuzzy matching to find the best match for the champion name if the input is not exact.
28/// - The message displaying the champion's information is automatically deleted after 60 seconds to keep the chat clean.
29///
30/// # Example:
31/// ```rust
32/// championsinfos(ctx).await?;
33/// ```
34///
35/// This command displays information such as:
36/// ```text
37/// 📝 Informations sur Jhin
38/// Rôles: AD Carry
39/// Winrate: 51.74%
40/// Banrate: 17.61%
41/// Popularité: 27.57%
42///
43/// Runes:
44/// **Rune principale:** <FleetFootwork Emoji>
45/// <PresenceOfMind Emoji> <LegendBloodline Emoji> <CoupDeGrace Emoji>
46///
47/// **Runes secondaires:** <Celerity Emoji>
48/// <GatheringStorm Emoji>
49///
50/// **Fragments:** <AdaptiveForce Emoji> <AdaptiveForce Emoji> <HealthScale Emoji>
51///
52/// Build d'objets:
53/// <StatikkShiv Emoji> <RapidFirecannon Emoji> <InfinityEdge Emoji>
54/// ```
55///
56/// # Errors:
57/// - If the user does not provide any input in the modal.
58/// - If the champion is not found in the database.
59/// - If there is an error retrieving data from the database or during any asynchronous operation.
60///
61/// # See Also:
62/// - `create_embed_champions_info`: Constructs the embed with the champion's information.
63/// - `ChampionsInfosModal`: The modal dialog used to collect the champion's name from the user.
64///
65/// # Related Commands:
66/// - `lolstats`: Fetches and displays LoL player stats based on user input.
67#[poise::command(slash_command)]
68pub async fn championsinfos(ctx: poise::ApplicationContext<'_, Data, Error>) -> Result<(), Error> {
69    let modal_data: ChampionsInfosModal = match ChampionsInfosModal::execute(ctx).await {
70        Ok(Some(data)) => data,
71        Ok(None) => {
72            let error_message = "Aucune donnée n'a été entrée.";
73            let reply = ctx.send(create_embed_error(&error_message)).await?;
74            schedule_message_deletion(reply, ctx).await?;
75            return Ok(());
76        }
77        Err(_) => {
78            let error_message = "Échec de la récupération des données du modal.";
79            let reply = ctx.send(create_embed_error(&error_message)).await?;
80            schedule_message_deletion(reply, ctx).await?;
81            return Ok(());
82        }
83    };
84
85    manage_user(
86        ctx.author().id.to_string(),
87        ctx.author().name.clone(),
88        &ctx.data().mongo_client,
89        false,
90    )
91    .await?;
92
93    let input_name = modal_data.champion_name.trim().to_lowercase();
94    let dd_json = &*ctx.data().dd_json.read().await;
95    let champion_names = get_champion_names(dd_json);
96    if champion_names.is_empty() {
97        let error_message = "Impossible de récupérer la liste des champions.";
98        let reply = ctx.send(create_embed_error(&error_message)).await?;
99        schedule_message_deletion(reply, ctx).await?;
100        return Ok(());
101    }
102
103    let matched_champion = champion_names
104        .iter()
105        .max_by(|a, b| {
106            let score_a = normalized_levenshtein(&input_name, &a.to_lowercase());
107            let score_b = normalized_levenshtein(&input_name, &b.to_lowercase());
108            score_a
109                .partial_cmp(&score_b)
110                .unwrap_or(std::cmp::Ordering::Equal)
111        })
112        .unwrap();
113    let matched_champion_id = get_champion_id(dd_json, matched_champion).unwrap();
114
115    let mongo_client: &mongodb::Client = &ctx.data().mongo_client;
116    let filter = doc! { "id_name": matched_champion_id};
117    let collection_champions = mongo_client
118        .database("stat-summoner")
119        .collection::<ChampionData>("champions_data");
120    let collection_emoji = mongo_client
121        .database("stat-summoner")
122        .collection::<EmojiId>("emojis_id");
123    match collection_champions.find_one(filter).await {
124        Ok(Some(champion_data)) => {
125            let embed = create_embed_champions_info(champion_data, &collection_emoji).await?;
126            let reply = CreateReply {
127                embeds: vec![embed],
128                ..Default::default()
129            };
130            let sent_message = ctx.send(reply).await?;
131            if let Err(e) = schedule_message_deletion(sent_message, ctx).await {
132                log::error!("Failed to schedule message deletion: {}", e);
133            }
134        }
135        Ok(None) => return Ok(()),
136        Err(e) => {
137            log::error!("Erreur lors de la recherche de l'emoji: {:?}", e);
138            return Ok(());
139        }
140    }
141
142    Ok(())
143}
